    fn collect_local_vars_in_subtree(node: Node, src: &[u8], out: &mut HashSet<String>) {
        if node.kind() == "declaration" {
            // Skip declarations with "static" or "extern" storage class
            let has_static_or_extern = {
                let mut cur = node.walk();
                node.children(&mut cur).any(|c| {
                    c.kind() == "storage_class_specifier"
                        && matches!(c.utf8_text(src).unwrap_or(""), "static" | "extern")
                })
            };
            if !has_static_or_extern {
                // Walk the declaration's declarator(s) to extract identifier names.
                // Handles: simple identifier, init_declarator → identifier, array_declarator → identifier, pointer_declarator → identifier.
                collect_decl_identifiers(node, src, out);
            }
        }
        let mut cursor = node.walk(); // Pitfall 1
        if cursor.goto_first_child() {
            loop {
                collect_local_vars_in_subtree(cursor.node(), src, out);
                if !cursor.goto_next_sibling() { break; }
            }
        }
    }

    fn collect_decl_identifiers(node: Node, src: &[u8], out: &mut HashSet<String>) {
        if node.kind() == "identifier" {
            if let Ok(name) = node.utf8_text(src) {
                out.insert(name.to_string());
            }
            return;
        }
        let mut cursor = node.walk();
        if cursor.goto_first_child() {
            loop {
                collect_decl_identifiers(cursor.node(), src, out);
                if !cursor.goto_next_sibling() { break; }
            }
        }
    }
    ```

    fn is_large_hex_literal(node: Node, src: &[u8]) -> bool {
        if node.kind() == "number_literal" {
            if let Ok(text) = node.utf8_text(src) {
                let trimmed = text.trim_end_matches(|c: char| matches!(c, 'u' | 'U' | 'l' | 'L'));
                if trimmed.starts_with("0x") || trimmed.starts_with("0X") {
                    if let Ok(val) = u64::from_str_radix(&trimmed[2..], 16) {
                        return val > 0xFFFF;
                    }
                }
            }
        }
        false
    }

    fn check_fixed_address_assignment(
        node: Node,
        src: &[u8],
        path: &Path,
        component_name: &str,
        component_ecosystem: &str,
        findings: &mut Vec<SastFinding>,
    ) {
        // Fire on init_declarator whose value child is cast_expression wrapping a large hex literal,
        // OR cast_expression of large hex literal in assignment_expression RHS.
        let value_node = if node.kind() == "init_declarator" {
            node.child_by_field_name("value")
        } else if node.kind() == "assignment_expression" {
            node.child_by_field_name("right")
        } else {
            None
        };

        if let Some(v) = value_node {
            // Unwrap cast_expression to get its value child (the literal).
            let candidate = if v.kind() == "cast_expression" {
                v.child_by_field_name("value").unwrap_or(v)
            } else {
                v
            };
            if is_large_hex_literal(candidate, src) {
                findings.push(SastFinding {
                    cwe_id: 587,
                    component_name: component_name.to_string(),
                    component_ecosystem: component_ecosystem.to_string(),
                    file_path: path.to_string_lossy().into_owned(),
                    line: (node.start_position().row as u32) + 1,
                    source: SastSource::Ast,
                });
            }
        }

        let mut cursor = node.walk(); // Pitfall 1
        if cursor.goto_first_child() {
            loop {
                check_fixed_address_assignment(cursor.node(), src, path, component_name, component_ecosystem, findings);
                if !cursor.goto_next_sibling() { break; }
            }
        }
    }
    ```
