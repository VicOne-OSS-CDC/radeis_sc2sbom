<ground_truth_format>
Ground truth file is a UTF-8 plaintext file at `<fixture>/.benchmark_truth.tsv` (TSV) with the following columns:
```
relative_file_path\tline_number\tcwe_id\tlabel
```
where `label` is `TP` (true positive — this is a real vulnerability) or `FP` (this would be a false positive if any scanner reports it).

A finding produced by any scanner is classified as:
- **TP** if `(relative_path, line, cwe_id)` matches a row with label `TP`
- **FP** if `(relative_path, line, cwe_id)` matches a row with label `FP`, OR if no matching row exists AND the file is listed in a `<fixture>/.benchmark_files.txt` file (one path per line) indicating that this file has been audited (otherwise the finding is `unaudited` and excluded from the table totals)

If `.benchmark_truth.tsv` is missing, the benchmark writes raw finding counts only and notes the missing oracle in BENCHMARK.md.
</ground_truth_format>

<file_layout>
**docs/BENCHMARK.md file layout (run-order independent):**

The file is organized as:
1. Top-of-file header block: title, "Generated by ..." timestamp, columns legend. This block uses fixed marker strings (`<!-- BENCHMARK-HEADER-START -->` ... `<!-- BENCHMARK-HEADER-END -->`) so the writer can detect and refresh it idempotently.
2. Zero or more fixture sections, each delimited by `<!-- FIXTURE-SECTION-START: {fixture_name} -->` ... `<!-- FIXTURE-SECTION-END: {fixture_name} -->` markers.

Writer algorithm (`write_benchmark_md`):
- Read the current `docs/BENCHMARK.md` (empty string if absent).
- If the header block is missing, prepend the header block (with current UTC timestamp).
- For this fixture: if a section with matching `{fixture_name}` markers exists, replace it in-place; otherwise append a new section before EOF.
- Write the result back.

This makes each `#[test]` function self-contained: running only AUTOSAR produces a valid file with just AUTOSAR's section; running only Juliet produces a valid file with just Juliet's section; running both (in either order) produces a file with both sections, each updated to the latest values from that test run.
</file_layout>

    ```rust
    #![cfg(feature = "internal")]
    //! Phase 18 benchmark integration test (D-10..D-14, BENCH-01).
    //!
    //! Not CI-gated. Runs locally when fixtures are present:
    //!   AUTOSAR_FIXTURE_PATH (or `../AUTOSAR_SampleProject_S32K144`)
    //!   JULIET_FIXTURE_PATH  (or `../juliet_c_cpp_subset`)
    //!
    //! Writes per-fixture, per-CWE TP/FP comparison to docs/BENCHMARK.md.
    //! Each test writes a self-contained fixture section using HTML-comment markers,
    //! so the file is correct regardless of test execution order or partial runs.
    //! See docs/BENCHMARK_FIXTURES.md for fixture acquisition and ground-truth format.

    use std::collections::{BTreeMap, BTreeSet, HashMap};
    use std::path::{Path, PathBuf};
    use radeis_sc2sbom::vulnerability::{
        run_ast_scanner, run_cppcheck_scanner, run_lexical_scanner,
        SastFinding,
    };

    const HEADER_START: &str = "<!-- BENCHMARK-HEADER-START -->";
    const HEADER_END:   &str = "<!-- BENCHMARK-HEADER-END -->";

    fn section_start(name: &str) -> String { format!("<!-- FIXTURE-SECTION-START: {} -->", name) }
    fn section_end(name: &str)   -> String { format!("<!-- FIXTURE-SECTION-END: {} -->",   name) }

    fn fixture_path(env_var: &str, default: &str) -> Option<PathBuf> {
        let path = std::env::var(env_var).unwrap_or_else(|_| default.to_string());
        let p = PathBuf::from(&path);
        if p.exists() { Some(p) } else { None }
    }

    fn load_ground_truth(fixture: &Path) -> Option<HashMap<(String, u32, u32), String>> {
        let tsv = fixture.join(".benchmark_truth.tsv");
        if !tsv.exists() { return None; }
        let body = std::fs::read_to_string(&tsv).ok()?;
        let mut map = HashMap::new();
        for line in body.lines() {
            if line.starts_with('#') || line.trim().is_empty() { continue; }
            let cols: Vec<&str> = line.split('\t').collect();
            if cols.len() < 4 { continue; }
            let rel = cols[0].to_string();
            let line_no: u32 = cols[1].parse().ok()?;
            let cwe: u32 = cols[2].parse().ok()?;
            let label = cols[3].trim().to_string();
            if label != "TP" && label != "FP" { continue; }
            map.insert((rel, line_no, cwe), label);
        }
        Some(map)
    }

    fn classify(
        finding: &SastFinding,
        fixture: &Path,
        truth: Option<&HashMap<(String, u32, u32), String>>,
    ) -> Option<&'static str> {
        let truth = truth?;
        let rel = Path::new(&finding.file_path)
            .strip_prefix(fixture)
            .ok()
            .map(|p| p.to_string_lossy().into_owned())
            .unwrap_or_else(|| finding.file_path.clone());
        match truth.get(&(rel, finding.line, finding.cwe_id)).map(|s| s.as_str()) {
            Some("TP") => Some("TP"),
            Some("FP") => Some("FP"),
            _ => None, // unaudited
        }
    }

    fn build_component_dirs(fixture: &Path) -> HashMap<(String, String), PathBuf> {
        let mut m = HashMap::new();
        m.insert(
            ("fixture".to_string(), "makefile".to_string()),
            fixture.to_path_buf(),
        );
        m
    }

    fn locate_cppcheck() -> Option<PathBuf> {
        if let Ok(path) = std::env::var("CPPCHECK_BIN") {
            let p = PathBuf::from(path);
            if p.exists() { return Some(p); }
        }
        let output = std::process::Command::new("which")
            .arg("cppcheck")
            .output()
            .ok()?;
        if !output.status.success() { return None; }
        let p = PathBuf::from(String::from_utf8(output.stdout).ok()?.trim());
        if p.exists() { Some(p) } else { None }
    }

    struct ScannerCounts {
        tp: u32,
        fp: u32,
        total: u32, // including unaudited
    }

    fn count_by_cwe(
        findings: &[SastFinding],
        fixture: &Path,
        truth: Option<&HashMap<(String, u32, u32), String>>,
    ) -> BTreeMap<u32, ScannerCounts> {
        let mut by_cwe: BTreeMap<u32, ScannerCounts> = BTreeMap::new();
        for f in findings {
            let e = by_cwe.entry(f.cwe_id).or_insert(ScannerCounts { tp: 0, fp: 0, total: 0 });
            e.total += 1;
            match classify(f, fixture, truth) {
                Some("TP") => e.tp += 1,
                Some("FP") => e.fp += 1,
                _ => {}
            }
        }
        by_cwe
    }

    /// Render the top-of-file header block (between HEADER_START / HEADER_END markers).
    fn render_header() -> String {
        let mut s = String::new();
        s.push_str(HEADER_START);
        s.push('\n');
        s.push_str("# Phase 18 Benchmark — AST vs cppcheck vs Lexical\n\n");
        s.push_str(&format!(
            "Generated by `cargo test --features internal --test benchmark` on {}.\n\n",
            chrono::Utc::now().format("%Y-%m-%d %H:%M UTC")
        ));
        s.push_str("Columns: AST TPs | AST FPs | AST FP% | cppcheck TPs | cppcheck FPs | cppcheck FP% | Lexical TPs | Lexical FPs | Lexical FP% (per D-14).\n\n");
        s.push_str(HEADER_END);
        s.push('\n');
        s
    }

    /// Render a single fixture section (between its FIXTURE-SECTION-START/END markers).
    fn render_fixture_section(
        fixture_name: &str,
        fixture_path: &Path,
        truth_present: bool,
        ast: &BTreeMap<u32, ScannerCounts>,
        cppcheck: Option<&BTreeMap<u32, ScannerCounts>>,
        lexical: &BTreeMap<u32, ScannerCounts>,
    ) -> String {
        let mut s = String::new();
        s.push_str(&section_start(fixture_name));
        s.push('\n');
        s.push_str(&format!("## Fixture: {}\n\n", fixture_name));
        s.push_str(&format!("Path: `{}`\n\n", fixture_path.display()));
        if !truth_present {
            s.push_str("**Ground-truth file `.benchmark_truth.tsv` not present — only raw finding counts are reported (no TP/FP/FP%).**\n\n");
        }
        let mut all_cwes: BTreeSet<u32> = BTreeSet::new();
        all_cwes.extend(ast.keys().copied());
        all_cwes.extend(lexical.keys().copied());
        if let Some(c) = cppcheck { all_cwes.extend(c.keys().copied()); }

        s.push_str("| CWE ID | AST TPs | AST FPs | AST FP% | cppcheck TPs | cppcheck FPs | cppcheck FP% | Lexical TPs | Lexical FPs | Lexical FP% |\n");
        s.push_str("|--------|---------|---------|---------|---------------|---------------|----------------|--------------|--------------|----------------|\n");
        let fp_pct = |c: &ScannerCounts| -> String {
            let denom = c.tp + c.fp;
            if denom == 0 { "—".to_string() } else { format!("{:.1}%", 100.0 * c.fp as f64 / denom as f64) }
        };
        let z = ScannerCounts { tp: 0, fp: 0, total: 0 };
        let empty: BTreeMap<u32, ScannerCounts> = BTreeMap::new();
        let cpp = cppcheck.unwrap_or(&empty);
        for cwe in &all_cwes {
            let a = ast.get(cwe).unwrap_or(&z);
            let c = cpp.get(cwe).unwrap_or(&z);
            let l = lexical.get(cwe).unwrap_or(&z);
            let cpp_cell = if cppcheck.is_none() {
                "n/a | n/a | n/a".to_string()
            } else {
                format!("{} | {} | {}", c.tp, c.fp, fp_pct(c))
            };
            s.push_str(&format!(
                "| CWE-{} | {} | {} | {} | {} | {} | {} | {} |\n",
                cwe, a.tp, a.fp, fp_pct(a), cpp_cell, l.tp, l.fp, fp_pct(l)
            ));
        }
        s.push('\n');
        s.push_str(&section_end(fixture_name));
        s.push('\n');
        s
    }

    /// Idempotent, run-order-independent writer.
    /// - Ensures the top-of-file header block exists (refreshes its timestamp on each write).
    /// - Replaces the named fixture section in place if present; appends it otherwise.
    fn write_benchmark_md(
        fixture_name: &str,
        fixture_path: &Path,
        truth_present: bool,
        ast: &BTreeMap<u32, ScannerCounts>,
        cppcheck: Option<&BTreeMap<u32, ScannerCounts>>,
        lexical: &BTreeMap<u32, ScannerCounts>,
    ) {
        let out_path = Path::new(env!("CARGO_MANIFEST_DIR")).join("docs/BENCHMARK.md");
        if let Some(parent) = out_path.parent() {
            let _ = std::fs::create_dir_all(parent);
        }

        let existing = std::fs::read_to_string(&out_path).unwrap_or_default();
        let new_header = render_header();
        let new_section = render_fixture_section(
            fixture_name, fixture_path, truth_present, ast, cppcheck, lexical,
        );

        // Step 1: replace or insert the header block.
        let after_header: String = if let (Some(s), Some(e)) = (
            existing.find(HEADER_START),
            existing.find(HEADER_END),
        ) {
            let e_end = e + HEADER_END.len();
            // Replace [s..e_end] with new_header (trim trailing newline of new_header to avoid double).
            let before = &existing[..s];
            let after  = &existing[e_end..];
            let after_trimmed = after.trim_start_matches('\n');
            format!("{}{}{}", before, new_header, after_trimmed)
        } else {
            // Prepend.
            format!("{}{}", new_header, existing)
        };

        // Step 2: replace or append the named fixture section.
        let s_marker = section_start(fixture_name);
        let e_marker = section_end(fixture_name);
        let final_content: String = if let (Some(s), Some(e)) = (
            after_header.find(&s_marker),
            after_header.find(&e_marker),
        ) {
            let e_end = e + e_marker.len();
            let before = &after_header[..s];
            let after  = &after_header[e_end..];
            let after_trimmed = after.trim_start_matches('\n');
            format!("{}{}{}", before, new_section, after_trimmed)
        } else {
            // Append (ensure trailing newline separation).
            let mut s = after_header;
            if !s.ends_with('\n') { s.push('\n'); }
            s.push_str(&new_section);
            s
        };

        std::fs::write(&out_path, final_content).expect("failed to write docs/BENCHMARK.md");
    }

    fn run_one_fixture(fixture_name: &str, fixture: &Path) {
        let component_dirs = build_component_dirs(fixture);
        let truth = load_ground_truth(fixture);

        eprintln!("Benchmark[{}]: running AST scanner...", fixture_name);
        let ast_findings = run_ast_scanner(&component_dirs);
        eprintln!("Benchmark[{}]: AST produced {} findings", fixture_name, ast_findings.len());

        eprintln!("Benchmark[{}]: running lexical scanner...", fixture_name);
        let lexical_findings = run_lexical_scanner(&component_dirs);
        eprintln!("Benchmark[{}]: lexical produced {} findings", fixture_name, lexical_findings.len());

        let cppcheck_bin = locate_cppcheck();
        let cppcheck_findings = match &cppcheck_bin {
            Some(p) => {
                eprintln!("Benchmark[{}]: running cppcheck ({:?})...", fixture_name, p);
                let (f, _dirs) = run_cppcheck_scanner(&component_dirs, Some(p.as_path()));
                eprintln!("Benchmark[{}]: cppcheck produced {} findings", fixture_name, f.len());
                Some(f)
            }
            None => {
                eprintln!("Benchmark[{}]: cppcheck NOT FOUND on PATH — cppcheck columns will be n/a", fixture_name);
                None
            }
        };

        let ast_counts = count_by_cwe(&ast_findings, fixture, truth.as_ref());
        let lexical_counts = count_by_cwe(&lexical_findings, fixture, truth.as_ref());
        let cppcheck_counts = cppcheck_findings.as_ref().map(|f| count_by_cwe(f, fixture, truth.as_ref()));

        write_benchmark_md(
            fixture_name,
            fixture,
            truth.is_some(),
            &ast_counts,
            cppcheck_counts.as_ref(),
            &lexical_counts,
        );
        eprintln!("Benchmark[{}]: wrote docs/BENCHMARK.md", fixture_name);
    }
    ```
