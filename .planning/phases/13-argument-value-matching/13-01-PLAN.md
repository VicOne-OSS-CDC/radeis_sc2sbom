    ```rust
    /// Returns true iff the paren-bound argument list of a function call (slice
    /// starting at `(`) contains EVERY token in `tokens` with word-boundary
    /// matching. AND-all semantics per D-02.
    ///
    /// `after_func` may have leading whitespace; the function strips it and
    /// then expects '('. Tracks paren depth so nested calls in arguments
    /// (e.g. `f(x, callback(y))`) are handled correctly (D-04, Pitfall 4).
    fn paren_args_contain_all(after_func: &str, tokens: &[&str]) -> bool {
        let rest = after_func.trim_start();
        let inner = match rest.strip_prefix('(') {
            Some(r) => r,
            None => return false,
        };
        let mut depth: i32 = 0;
        let mut end = inner.len();
        for (i, ch) in inner.char_indices() {
            match ch {
                '(' => depth += 1,
                ')' => {
                    if depth == 0 {
                        end = i;
                        break;
                    }
                    depth -= 1;
                }
                _ => {}
            }
        }
        let arg_slice = &inner[..end];
        for &token in tokens {
            if !token_present_with_boundary(arg_slice, token) {
                return false;
            }
        }
        true
    }

    /// Word-boundary substring search. Left side: prev char must not be
    /// alnum or `_`. Right side: next char must not be alnum or `_`,
    /// AND for all-digit tokens (D-05) must also not be a digit or `.`
    /// (so `umask(0077)` and `x / 0.5` do NOT match token `"0"`).
    fn token_present_with_boundary(haystack: &str, token: &str) -> bool {
        if token.is_empty() {
            return false;
        }
        let bytes = haystack.as_bytes();
        let tok_len = token.len();
        let is_digit_token = token.bytes().all(|b| b.is_ascii_digit());
        let mut i = 0usize;
        while i <= haystack.len().saturating_sub(tok_len) {
            let rel = match haystack[i..].find(token) {
                Some(r) => r,
                None => return false,
            };
            let pos = i + rel;
            let left_ok = if pos == 0 {
                true
            } else {
                let prev = bytes[pos - 1];
                !(prev.is_ascii_alphanumeric() || prev == b'_')
            };
            let after_pos = pos + tok_len;
            let right_ok = if after_pos >= bytes.len() {
                true
            } else {
                let next = bytes[after_pos];
                if is_digit_token {
                    !(next.is_ascii_digit() || next == b'.')
                } else {
                    !(next.is_ascii_alphanumeric() || next == b'_')
                }
            };
            if left_ok && right_ok {
                return true;
            }
            i = pos + 1;
        }
        false
    }
    ```

    ```rust
    /// CWE-369 separate scan path (D-09). Returns true iff `line` contains
    /// `/` or `%` followed by optional whitespace and a standalone `0`
    /// literal (right boundary: next char must not be a digit or `.`,
    /// so `0.5`, `0x10`, `0077`, `10` do NOT match).
    ///
    /// Note: lexical-only — does not strip C/C++ comments. A line like
    /// `// divide by 0` will produce a finding. This matches the
    /// existing scanner's design philosophy (Pitfall 3 in 13-RESEARCH.md).
    fn contains_div_by_zero(line: &str) -> bool {
        let bytes = line.as_bytes();
        let len = bytes.len();
        let mut i = 0usize;
        while i < len {
            if bytes[i] == b'/' || bytes[i] == b'%' {
                let mut j = i + 1;
                while j < len && (bytes[j] == b' ' || bytes[j] == b'\t') {
                    j += 1;
                }
                if j < len && bytes[j] == b'0' {
                    let after = j + 1;
                    let right_ok = after >= len
                        || (!bytes[after].is_ascii_digit() && bytes[after] != b'.');
                    if right_ok {
                        return true;
                    }
                }
            }
            i += 1;
        }
        false
    }
    ```
