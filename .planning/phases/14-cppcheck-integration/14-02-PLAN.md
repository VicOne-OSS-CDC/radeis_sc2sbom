  ```rust
  /// Parse cppcheck XML v2 output (the bytes cppcheck writes to stderr when
  /// invoked with `--xml --xml-version=2`) into a list of SastFinding entries.
  ///
  /// CWE resolution order (D-01):
  ///   1. The `cwe` attribute on `<error>`.
  ///   2. CPPCHECK_CWE_OVERRIDES lookup by `id` attribute when (1) is absent or 0.
  /// Findings with no resolvable CWE are silently dropped (D-04).
  /// Only the FIRST `<location>` child of an `<error>` is used (D-05/Pattern G).
  pub(crate) fn parse_cppcheck_xml(
      xml_bytes: &[u8],
      component_name: &str,
      component_ecosystem: &str,
  ) -> Vec<SastFinding> {
      let mut reader = Reader::from_reader(xml_bytes);
      reader.config_mut().trim_text(true);
      let mut buf = Vec::new();
      let mut findings = Vec::new();

      let mut current_id = String::new();
      let mut current_cwe: u32 = 0;
      let mut location_taken = false;

      loop {
          match reader.read_event_into(&mut buf) {
              Ok(Event::Start(ref e)) | Ok(Event::Empty(ref e))
                  if e.name().as_ref() == b"error" =>
              {
                  current_id.clear();
                  current_cwe = 0;
                  location_taken = false;
                  for attr in e.attributes().flatten() {
                      match attr.key.as_ref() {
                          b"id" => {
                              current_id =
                                  String::from_utf8_lossy(&attr.value).into_owned();
                          }
                          b"cwe" => {
                              current_cwe = String::from_utf8_lossy(&attr.value)
                                  .parse::<u32>()
                                  .unwrap_or(0);
                          }
                          _ => {}
                      }
                  }
                  if current_cwe == 0 {
                      for &(id, cwe) in CPPCHECK_CWE_OVERRIDES {
                          if id == current_id.as_str() {
                              current_cwe = cwe;
                              break;
                          }
                      }
                  }
              }
              Ok(Event::Empty(ref e))
                  if e.name().as_ref() == b"location"
                      && !location_taken
                      && current_cwe != 0 =>
              {
                  let mut file = String::new();
                  let mut line: u32 = 0;
                  for attr in e.attributes().flatten() {
                      match attr.key.as_ref() {
                          b"file" => {
                              file = String::from_utf8_lossy(&attr.value).into_owned();
                          }
                          b"line" => {
                              line = String::from_utf8_lossy(&attr.value)
                                  .parse::<u32>()
                                  .unwrap_or(0);
                          }
                          _ => {}
                      }
                  }
                  if !file.is_empty() && line > 0 {
                      findings.push(SastFinding {
                          cwe_id: current_cwe,
                          component_name: component_name.to_string(),
                          component_ecosystem: component_ecosystem.to_string(),
                          file_path: file,
                          line,
                          source: SastSource::Cppcheck,
                      });
                      location_taken = true;
                  }
              }
              Ok(Event::Eof) => break,
              Err(_) => break,
              _ => {}
          }
          buf.clear();
      }
      findings
  }
  ```

  Do NOT add `pub` (full pub) on the function; `pub(crate)` is sufficient — Plan 04 callers are in the same crate. Do NOT change `SastFinding`, `CWE_RULES`, or `run_lexical_scanner`.
