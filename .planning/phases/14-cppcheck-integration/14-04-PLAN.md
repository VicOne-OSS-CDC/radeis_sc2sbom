  ```rust
  /// Invoke cppcheck on every component-mapped C/C++ directory and return
  /// the union of all SastFindings produced.
  ///
  /// `cppcheck_bin` lets the caller override the PATH-resolved `cppcheck`
  /// binary (D-15, --cppcheck-path). When None, `cppcheck` is looked up via
  /// PATH by `Command::new`.
  ///
  /// Graceful degradation (D-09): if the preflight `--version` invocation
  /// fails, this function emits a single warning to stderr and returns
  /// Vec::new(). Per-component failures (D-10) are warned to stderr; the
  /// loop continues to the next component.
  pub fn run_cppcheck_scanner(
      component_dirs: &HashMap<(String, String), PathBuf>,
      cppcheck_bin: Option<&OsStr>,
  ) -> Vec<SastFinding> {
      // Resolve binary: explicit override or PATH lookup of "cppcheck".
      let bin: &OsStr = cppcheck_bin.unwrap_or_else(|| OsStr::new("cppcheck"));

      // Preflight (D-09): a successful --version invocation is the gate.
      let preflight = Command::new(bin).args(["--version"]).output();
      let ok = match &preflight {
          Ok(o) => o.status.success(),
          Err(_) => false,
      };
      if !ok {
          eprintln!(
              "⚠ cppcheck not found — lexical-only results. Install cppcheck or use --cppcheck-path."
          );
          return Vec::new();
      }

      // Spinner (D-07).
      let total = component_dirs.len();
      let pb = ProgressBar::new_spinner();
      let style = ProgressStyle::default_spinner()
          .tick_strings(&["⠋", "⠙", "⠹", "⠸", "⠼", "⠴", "⠦", "⠧", "⠇", "⠏"])
          .template("[cppcheck] Scanning {msg} {spinner}")
          .unwrap_or_else(|_| ProgressStyle::default_spinner());
      pb.set_style(style);
      pb.enable_steady_tick(std::time::Duration::from_millis(100));

      let mut all_findings: Vec<SastFinding> = Vec::new();
      let mut completed_components: usize = 0;

      for (i, ((name, ecosystem), dir)) in component_dirs.iter().enumerate() {
          pb.set_message(format!("{} ({}/{})", name, i + 1, total));
          if !dir.exists() {
              continue;
          }

          // D-05: invoke cppcheck once per component dir.
          // Pitfall 1: cppcheck writes XML to stderr, NOT stdout.
          let output = Command::new(bin)
              .args([
                  "--xml",
                  "--xml-version=2",
                  "--enable=warning,style,security",
                  dir.to_str().unwrap_or("."),
              ])
              .stdout(Stdio::null())
              .stderr(Stdio::piped())
              .output();

          match output {
              Err(e) => {
                  // D-10: warn and continue.
                  eprintln!("cppcheck: failed to spawn for {}: {}", name, e);
                  continue;
              }
              Ok(out) if !out.status.success() => {
                  // D-10: cppcheck non-zero exit; log component + stderr tail and continue.
                  eprintln!(
                      "cppcheck: non-zero exit for {}: {}",
                      name,
                      String::from_utf8_lossy(&out.stderr)
                  );
                  continue;
              }
              Ok(out) => {
                  let component_findings = parse_cppcheck_xml(&out.stderr, name, ecosystem);
                  all_findings.extend(component_findings);
                  completed_components += 1;
              }
          }
      }

      pb.finish_and_clear();
      // D-08: completion summary.
      eprintln!(
          "cppcheck: {} findings from {} components",
          all_findings.len(),
          completed_components
      );

      all_findings
  }
  ```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| sc2sbom → cppcheck subprocess | We spawn an external process. Args are statically assembled — only the directory path and the (override) binary path come from program state. |
| cppcheck → sc2sbom (stderr bytes) | Already covered by Plan 02 threat model; same parser path. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-14-08 | Elevation of privilege | `Command::new(bin)` with user-supplied path | accept | Flag is gated behind `#[cfg(feature = "internal")]` (T-14-06 in Plan 03 covers the rationale). The argv array is constructed with discrete strings (no shell), so command injection via the path is not possible. |
| T-14-09 | Denial of service | cppcheck never returns / runs indefinitely | accept | No timeout is implemented in this phase (CPPCHECK-F2 deferred per CONTEXT.md). Risk is local-only — affects only the user running their own scanner against their own code. Future enhancement when CI timing is in scope. |
| T-14-10 | Information disclosure | Component name/path leaked to stderr summary | accept | The summary line is `cppcheck: N findings from M components` — no paths leaked. Per-component error logs include the component name (intended diagnostic) but not external secrets. |
| T-14-11 | Tampering | dir.to_str() lossy conversion drops non-UTF8 paths | mitigate | We pass `dir.to_str().unwrap_or(".")` — if the path is non-UTF8, cppcheck scans `.` instead. This is a graceful degradation that still produces useful output for the common case; non-UTF8 component dirs are vanishingly rare on supported platforms (Linux/macOS UTF-8 default). Logged via spurious findings if it happens, but never panics. |
</threat_model>
