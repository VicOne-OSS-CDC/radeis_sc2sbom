    ```rust
    fn collect_function_scope_pointer_declarators(fn_node: Node, src: &[u8]) -> HashSet<String> {
        let mut result = HashSet::new();
        collect_ptrs_in_subtree(fn_node, src, &mut result);

    /// Collect pointer-typed variable names declared at file scope (NOT inside any function_definition).
    fn collect_file_scope_pointer_declarators(root: Node, src: &[u8]) -> HashSet<String> {
        let mut result = HashSet::new();
        collect_file_scope_ptrs_rec(root, src, &mut result);
        result
    }

    fn collect_file_scope_ptrs_rec(node: Node, src: &[u8], out: &mut HashSet<String>) {
        // Skip function_definition subtrees (those are local scope).
        if node.kind() == "function_definition" {
            return;
        }
        if node.kind() == "declaration" {
            collect_pointer_declarators(node, src, out);
        }
        let mut cursor = node.walk(); // fresh cursor per level (Pitfall 1)
        if cursor.goto_first_child() {
            loop {
                collect_file_scope_ptrs_rec(cursor.node(), src, out);
                if !cursor.goto_next_sibling() { break; }
            }
        }
    }

    fn collect_ptrs_in_subtree(node: Node, src: &[u8], out: &mut HashSet<String>) {
        if node.kind() == "declaration" {
            collect_pointer_declarators(node, src, out);
        }
        let mut cursor = node.walk(); // fresh cursor per level (Pitfall 1)
        if cursor.goto_first_child() {
            loop {
                collect_ptrs_in_subtree(cursor.node(), src, out);
                if !cursor.goto_next_sibling() { break; }
            }
        }
    }

    /// Scan a `declaration` node for `pointer_declarator` children and extract
    /// the underlying identifier name. Mirrors collect_array_declarators().
    fn collect_pointer_declarators(node: Node, src: &[u8], out: &mut HashSet<String>) {
        for i in 0..node.child_count() {
            if let Some(child) = node.child(i) {
                collect_pointer_declarator_rec(child, src, out);
            }
        }
    }

    fn collect_pointer_declarator_rec(node: Node, src: &[u8], out: &mut HashSet<String>) {
        if node.kind() == "pointer_declarator" {
            // tree-sitter-c: pointer_declarator's named child is the inner declarator.
            // Walk down to the first identifier (handles `char *p`, `char **pp`, init_declarator).
            extract_ident_under(node, src, out);
        }
        for i in 0..node.child_count() {
            if let Some(child) = node.child(i) {
                if child.kind() != "pointer_declarator" {
                    collect_pointer_declarator_rec(child, src, out);
                }
            }
        }
    }

    fn extract_ident_under(node: Node, src: &[u8], out: &mut HashSet<String>) {
        if node.kind() == "identifier" {
            if let Ok(name) = node.utf8_text(src) {
                out.insert(name.to_string());
            }
            return;
        }
        for i in 0..node.child_count() {
            if let Some(child) = node.child(i) {
                extract_ident_under(child, src, out);
            }
        }
    }
    ```

    ```rust
    /// Phase 22's CWE-480/481/482 wrong-operator detection.
    fn apply_division_rules(
        root: Node,
        src: &[u8],
        path: &Path,
        component_name: &str,
        component_ecosystem: &str,
        findings: &mut Vec<SastFinding>,
    ) {
        visit_binary_exprs(root, src, path, component_name, component_ecosystem, findings);
    }

    fn visit_binary_exprs<'a>(
        node: Node<'a>,
        src: &[u8],
        path: &Path,
        component_name: &str,
        component_ecosystem: &str,
        findings: &mut Vec<SastFinding>,
    ) {
        if node.kind() == "binary_expression" {
            // tree-sitter-c: operator is the unnamed token between left/right.
            // Access via child(1). Verify text via utf8_text.
            if let Some(op_node) = node.child(1) {
                if let Ok(op) = op_node.utf8_text(src) {
                    if op == "/" || op == "%" {
                        if let Some(rhs) = node.child_by_field_name("right") {
                            // Accept both "number_literal" and "integer_literal"
                            // for tree-sitter-c grammar version variance (RESEARCH §Pitfall 2).
                            let kind = rhs.kind();
                            if kind == "number_literal" || kind == "integer_literal" {
                                if let Ok(text) = rhs.utf8_text(src) {
                                    if text == "0" {
                                        findings.push(SastFinding {
                                            cwe_id: 369,
                                            component_name: component_name.to_string(),
                                            component_ecosystem: component_ecosystem.to_string(),
                                            file_path: path.to_string_lossy().into_owned(),
                                            line: (node.start_position().row as u32) + 1,
                                            source: SastSource::Ast,
                                        });
                                    }
                                }
                            }
                        }
                    }
                }
            }
        }
        // Recurse — fresh cursor per level (Pitfall 1).
        let mut cursor = node.walk();
        if cursor.goto_first_child() {
            loop {
                visit_binary_exprs(
                    cursor.node(), src, path, component_name, component_ecosystem, findings,
                );
                if !cursor.goto_next_sibling() { break; }
            }
        }
    }
    ```
