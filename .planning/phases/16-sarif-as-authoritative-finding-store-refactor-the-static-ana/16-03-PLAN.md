   Function body:
   ```rust
   /// SARIF-05: Write a diff SARIF containing only findings whose fingerprint is NOT
   /// in the baseline set. Returns the number of new findings.
   /// If count is 0, no file is written. If --sarif-output is supplied, write to that path;
   /// otherwise write to {out_dir}/{project_name}_static_analysis_diff.sarif.
   pub fn save_diff_sarif_report(
       project_name: &str,
       out_dir: &Path,
       findings: &[SastFinding],
       baseline_fingerprints: &std::collections::HashSet<String>,
       sarif_output: Option<&str>,
   ) -> Result<usize> {
       let new_findings: Vec<SastFinding> = findings
           .iter()
           .filter(|f| {
               let hash_key = sarif_fingerprint(&f.file_path, f.line, f.cwe_id);
               let tuple_key = format!("{}:{}:CWE-{}", f.file_path, f.line, f.cwe_id);
               !baseline_fingerprints.contains(&hash_key)
                   && !baseline_fingerprints.contains(&tuple_key)
           })
           .cloned()
           .collect();

       let count = new_findings.len();
       if count == 0 {
           return Ok(0);
       }

       let diff_path: PathBuf = match sarif_output {
           Some(p) => PathBuf::from(p),
           None => out_dir.join(format!("{}_static_analysis_diff.sarif", project_name)),
       };

       // Reuse the existing save_sarif_report pipeline by passing the diff_path explicitly.
       // save_sarif_report uses sarif_path.unwrap_or_else(...) for the output path; pass the
       // diff_path string so it writes to the correct location.
       let diff_path_str = diff_path.to_string_lossy().into_owned();
       save_sarif_report(project_name, out_dir, &new_findings, Some(&diff_path_str))?;

       Ok(count)
   }
   ```

   The trick of reusing `save_sarif_report` works because the existing function already accepts `sarif_path: Option<&str>` and writes to that path. This avoids duplicating the SarifLog/SarifRun/SarifTool/rules construction.

#![cfg(feature = "internal")]

use sc2sbom::formats::sarif::{save_diff_sarif_report, save_sarif_report, extract_baseline_fingerprints};
use sc2sbom::vulnerability::{SastFinding, SastSource};
use std::collections::HashSet;
use std::fs;

fn finding(cwe_id: u32, file_path: &str, line: u32) -> SastFinding {
    SastFinding {
        cwe_id,
        component_name: "lib".to_string(),
        component_ecosystem: "C/C++".to_string(),
        file_path: file_path.to_string(),
        line,
        source: SastSource::Lexical,
    }
}

#[test]
fn diff_returns_zero_when_all_findings_in_baseline() {
    let tmp = tempfile::tempdir().unwrap();
    let f = finding(120, "src/foo.c", 10);
    // Write baseline containing this finding.
    save_sarif_report("base", tmp.path(), &[f.clone()], None).unwrap();
    let baseline_path = tmp.path().join("base_static_analysis.sarif");
    let baseline_fps = extract_baseline_fingerprints(&baseline_path);

    let count = save_diff_sarif_report("cur", tmp.path(), &[f], &baseline_fps, None).unwrap();
    assert_eq!(count, 0);
    // No diff file should be written.
    assert!(!tmp.path().join("cur_static_analysis_diff.sarif").exists());
}

#[test]
fn diff_returns_count_for_new_findings() {
    let tmp = tempfile::tempdir().unwrap();
    let old = finding(120, "src/foo.c", 10);
    let new1 = finding(78, "src/bar.c", 20);
    let new2 = finding(190, "src/baz.c", 30);
    save_sarif_report("base", tmp.path(), &[old.clone()], None).unwrap();
    let baseline_fps = extract_baseline_fingerprints(&tmp.path().join("base_static_analysis.sarif"));

    let count = save_diff_sarif_report(
        "cur",
        tmp.path(),
        &[old, new1, new2],
        &baseline_fps,
        None,
    ).unwrap();
    assert_eq!(count, 2);
    let diff_path = tmp.path().join("cur_static_analysis_diff.sarif");
    assert!(diff_path.exists());
    let content = fs::read_to_string(&diff_path).unwrap();
    let json: serde_json::Value = serde_json::from_str(&content).unwrap();
    let results = json["runs"][0]["results"].as_array().unwrap();
    assert_eq!(results.len(), 2);
}

#[test]
fn diff_writes_to_sarif_output_when_provided() {
    let tmp = tempfile::tempdir().unwrap();
    let new = finding(78, "src/bar.c", 20);
    let baseline_fps: HashSet<String> = HashSet::new();
    let custom_path = tmp.path().join("custom_diff.sarif");
    let custom_str = custom_path.to_string_lossy().into_owned();

    let count = save_diff_sarif_report(

    assert_eq!(count, 1);

#![cfg(feature = "internal")]

use sc2sbom::formats::sarif::save_sarif_report;
use sc2sbom::formats::save_static_analysis_report;
use sc2sbom::vulnerability::{SastFinding, SastSource};
use std::fs;

fn finding(cwe_id: u32, file_path: &str, line: u32, source: SastSource) -> SastFinding {
    SastFinding {
        cwe_id,
        component_name: "lib".to_string(),
        component_ecosystem: "C/C++".to_string(),
        file_path: file_path.to_string(),
        line,
        source,
    }
}

#[test]
fn markdown_row_count_equals_sarif_results_length() {
    let tmp = tempfile::tempdir().unwrap();
    let findings = vec![
        finding(120, "src/a.c", 10, SastSource::Lexical),
        finding(78, "src/b.c", 20, SastSource::Cppcheck),
        finding(190, "src/c.c", 30, SastSource::Both),
    ];

    save_static_analysis_report("proj", tmp.path(), &findings).unwrap();
    save_sarif_report("proj", tmp.path(), &findings, None).unwrap();

    // Count SARIF results
    let sarif_content = fs::read_to_string(tmp.path().join("proj_static_analysis.sarif")).unwrap();
    let json: serde_json::Value = serde_json::from_str(&sarif_content).unwrap();
    let sarif_count = json["runs"][0]["results"].as_array().unwrap().len();

    // Count markdown finding rows. Strategy: read the file and count occurrences of "CWE-"
    // in finding rows. The existing markdown writer is the source of truth — count
    // occurrences of one of its per-finding line patterns. The test must work with
    // whatever the current markdown writer produces. Use a robust counter:
    // count distinct (file:line) appearances by counting "CWE-" occurrences and dividing
    // if the writer emits CWE- multiple times per row, OR — simpler — use a file:line
    // pattern that appears exactly once per finding.
    //
    // Read the file. For each finding, assert the file_path AND the line number AND
    // the cwe_id appear in the markdown content together at least once.
    let md_content = fs::read_to_string(tmp.path().join("proj_static_analysis.md")).unwrap();

    // Strict count: each finding's (file_path) must appear in the markdown.
    for f in &findings {
        assert!(md_content.contains(&f.file_path), "markdown missing {}", f.file_path);
    }

    // The invariant: SARIF results count == findings count (already true by construction).
    // The consistency assertion: markdown should contain every finding's signature.
    assert_eq!(sarif_count, findings.len(),
        "SARIF count {} != findings count {}", sarif_count, findings.len());

    // Counter for markdown rows: count how many of the (file_path, line, cwe_id) triples
    // appear together in the same line of the markdown content. This binds the markdown
    // count to SARIF count without depending on a specific table format.
    let lines: Vec<&str> = md_content.lines().collect();
    let mut md_finding_lines = 0;
    for f in &findings {
        let line_str = f.line.to_string();
        let cwe_str = format!("CWE-{}", f.cwe_id);
        for ln in &lines {
            if ln.contains(&f.file_path) && ln.contains(&line_str) && ln.contains(&cwe_str) {
                md_finding_lines += 1;
                break; // count each finding once
            }
        }
    }
    assert_eq!(md_finding_lines, sarif_count,
        "markdown rows ({}) != SARIF results ({})", md_finding_lines, sarif_count);
}
