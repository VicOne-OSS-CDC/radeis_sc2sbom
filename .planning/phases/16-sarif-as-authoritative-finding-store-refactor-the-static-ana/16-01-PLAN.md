      ```rust
      /// SARIF-07: Drop Lexical findings for CWEs in CPPCHECK_COVERED_CWES when
      /// cppcheck ran on the component dir and did NOT confirm the (file, line, cwe) site.
      ///
      /// `cppcheck_scanned_dirs` — directories where cppcheck successfully ran.
      /// `cppcheck_confirmed` — set of (normalized_file_path, line, cwe_id) tuples
      ///   that survived dedup with source == Cppcheck or Both.
      ///
      /// When cppcheck_scanned_dirs is empty (cppcheck did not run), input is returned unchanged.
      pub fn suppress_lexical_false_positives(
          findings: Vec<SastFinding>,
          cppcheck_scanned_dirs: &BTreeSet<PathBuf>,
          cppcheck_confirmed: &BTreeSet<(String, u32, u32)>,
      ) -> Vec<SastFinding> {
          if cppcheck_scanned_dirs.is_empty() {
              return findings;
          }
          findings.into_iter().filter(|f| {
              if f.source != SastSource::Lexical {
                  return true;
              }
              if !CPPCHECK_COVERED_CWES.contains(&f.cwe_id) {
                  return true;
              }
              let normalized = normalize_path(&f.file_path);
              let under_scanned_dir = cppcheck_scanned_dirs
                  .iter()
                  .any(|dir| Path::new(&normalized).starts_with(dir));
              if !under_scanned_dir {
                  return true;
              }
              let key = (normalized, f.line, f.cwe_id);
              cppcheck_confirmed.contains(&key)
          }).collect()
      }
      ```
