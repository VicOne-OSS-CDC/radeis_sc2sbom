2. Add the `extract_baseline_fingerprints` function at the bottom of the module (above any test module). Make it `pub` so Plan 03 / main.rs can call it via `formats::sarif::extract_baseline_fingerprints`:
   ```rust
   /// SARIF-05 helper (Phase 16): Load fingerprints from a baseline SARIF file.
   /// Returns an empty HashSet (with a warning to stderr) if the file is missing or
   /// malformed — never aborts. Falls back to `(uri, startLine, ruleId)` tuple keys
   /// for old SARIF baselines that lack `partialFingerprints`.
   pub fn extract_baseline_fingerprints(path: &Path) -> std::collections::HashSet<String> {
       let content = match std::fs::read_to_string(path) {
           Ok(c) => c,
           Err(e) => {
               eprintln!("Warning: could not read SARIF baseline {}: {}", path.display(), e);
               return std::collections::HashSet::new();
           }
       };
       let json: serde_json::Value = match serde_json::from_str(&content) {
           Ok(v) => v,
           Err(e) => {
               eprintln!("Warning: invalid SARIF baseline {}: {}", path.display(), e);
               return std::collections::HashSet::new();
           }
       };
       let mut fps = std::collections::HashSet::new();
       if let Some(results) = json["runs"][0]["results"].as_array() {
           for r in results {
               if let Some(fp) = r["partialFingerprints"]["primary/v1"].as_str() {
                   fps.insert(fp.to_string());
               } else {
                   let uri = r["locations"][0]["physicalLocation"]["artifactLocation"]["uri"]
                       .as_str().unwrap_or("");
                   let line = r["locations"][0]["physicalLocation"]["region"]["startLine"]
                       .as_u64().unwrap_or(0);
                   let rule = r["ruleId"].as_str().unwrap_or("");
                   fps.insert(format!("{}:{}:{}", uri, line, rule));
               }
           }
       }
       fps
   }
   ```
