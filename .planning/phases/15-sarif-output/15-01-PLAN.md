```rust
#![cfg(feature = "internal")]

use crate::vulnerability::cwe_scanner::SastFinding;
use anyhow::Result;
use serde::Serialize;
use std::collections::BTreeSet;
use std::fs;
use std::path::{Path, PathBuf};

use super::console::cwe_name;

#[derive(Serialize)]
struct SarifLog {
    #[serde(rename = "$schema")]
    schema: &'static str,
    version: &'static str,
    runs: Vec<SarifRun>,
}

#[derive(Serialize)]
struct SarifRun {
    tool: SarifTool,
    results: Vec<SarifResult>,
}

#[derive(Serialize)]
struct SarifTool {
    driver: SarifDriver,
}

#[derive(Serialize)]
struct SarifDriver {
    name: &'static str,
    version: &'static str,
    rules: Vec<SarifRule>,
}

#[derive(Serialize)]
#[serde(rename_all = "camelCase")]
struct SarifRule {
    id: String,
    name: String,
    help_uri: String,
}

#[derive(Serialize)]
#[serde(rename_all = "camelCase")]
struct SarifResult {
    rule_id: String,
    message: SarifMessage,
    locations: Vec<SarifLocation>,
}

#[derive(Serialize)]
struct SarifMessage {
    text: String,
}

#[derive(Serialize)]
#[serde(rename_all = "camelCase")]
struct SarifLocation {
    physical_location: SarifPhysicalLocation,
}

#[derive(Serialize)]
#[serde(rename_all = "camelCase")]
struct SarifPhysicalLocation {
    artifact_location: SarifArtifactLocation,
    region: SarifRegion,
}

#[derive(Serialize)]
struct SarifArtifactLocation {
    uri: String,
}

#[derive(Serialize)]
#[serde(rename_all = "camelCase")]
struct SarifRegion {
    start_line: u32,
}

/// Write SARIF 2.1 JSON file containing all SastFindings.
/// If sarif_path is Some, writes there; otherwise writes out_dir/{project_name}_static_analysis.sarif.
pub fn save_sarif_report(
    project_name: &str,
    out_dir: &Path,
    findings: &[SastFinding],
    sarif_path: Option<&str>,
) -> Result<()> {
    // Resolve output path (D-01, D-02)
    let path = match sarif_path {
        Some(p) => PathBuf::from(p),
        None => out_dir.join(format!("{}_static_analysis.sarif", project_name)),
    };

    // Create parent dir if needed (handles --sarif-output with non-existent parent)
    if let Some(parent) = path.parent() {
        if !parent.as_os_str().is_empty() {
            fs::create_dir_all(parent)?;
        }
    }

    // rules[]: deduplicated by CWE, deterministic order via BTreeSet (D-05)
    let unique_cwes: BTreeSet<u32> = findings.iter().map(|f| f.cwe_id).collect();
    let rules: Vec<SarifRule> = unique_cwes
        .iter()
        .map(|&id| SarifRule {
            id: format!("CWE-{}", id),
            name: cwe_name(id).to_string(),
            help_uri: format!("https://cwe.mitre.org/data/definitions/{}.html", id),
        })
        .collect();

    // results[]: one entry per finding, no dedup (CONTEXT.md <specifics>)
    let results: Vec<SarifResult> = findings
        .iter()
        .map(|f| SarifResult {
            rule_id: format!("CWE-{}", f.cwe_id),
            message: SarifMessage {
                text: cwe_name(f.cwe_id).to_string(),
            },
            locations: vec![SarifLocation {
                physical_location: SarifPhysicalLocation {
                    artifact_location: SarifArtifactLocation {
                        uri: f.file_path.clone(),
                    },
                    region: SarifRegion {
                        start_line: f.line,
                    },
                },
            }],
        })
        .collect();

    let log = SarifLog {
        schema: "https://raw.githubusercontent.com/oasis-tcs/sarif-spec/master/Schemata/sarif-schema-2.1.0.json",
        version: "2.1.0",
        runs: vec![SarifRun {
            tool: SarifTool {
                driver: SarifDriver {
                    name: "sc2sbom",
                    version: env!("CARGO_PKG_VERSION"),
                    rules,
                },
            },
            results,
        }],
    };

    let json = serde_json::to_string_pretty(&log)?;
    fs::write(&path, json)?;
    eprintln!("✓ SARIF report saved to: {}", path.display());
    Ok(())
}
```

Step A — Create tests/format_tests/sarif_tests.rs with the following exact contents:

```rust
#![cfg(feature = "internal")]

use radeis_sc2sbom::formats::save_sarif_report;
use radeis_sc2sbom::vulnerability::cwe_scanner::SastFinding;
use serde_json::Value;
use tempfile::TempDir;

fn make_finding(component: &str, file: &str, line: u32, cwe_id: u32) -> SastFinding {
    SastFinding {
        cwe_id,
        component_name: component.to_string(),
        component_ecosystem: "vendored".to_string(),
        file_path: file.to_string(),
        line,
    }
}

fn read_sarif(tmp: &TempDir, project_name: &str) -> Value {
    let path = tmp
        .path()
        .join(format!("{}_static_analysis.sarif", project_name));
    let s = std::fs::read_to_string(&path).expect("sarif file should exist");
    serde_json::from_str(&s).expect("sarif should be valid JSON")
}

#[test]
fn test_sarif_writes_default_path() {
    let tmp = TempDir::new().unwrap();
    let findings = vec![
        make_finding("libfoo", "src/libfoo/buffer.c", 42, 120),
        make_finding("libfoo", "src/libfoo/exec.c", 17, 78),
        make_finding("libbar", "src/libbar/utils.c", 99, 120),
    ];
    save_sarif_report("myproj", tmp.path(), &findings, None).unwrap();
    let path = tmp.path().join("myproj_static_analysis.sarif");
    assert!(path.exists(), "expected SARIF file at default path");
    let _: Value = read_sarif(&tmp, "myproj");
}

#[test]
fn test_sarif_schema_and_version() {
    let tmp = TempDir::new().unwrap();
    save_sarif_report("p", tmp.path(), &[], None).unwrap();
    let v = read_sarif(&tmp, "p");
    let schema = v["$schema"].as_str().unwrap();
    assert!(
        schema.starts_with("https://raw.githubusercontent.com/oasis-tcs/sarif-spec"),
        "schema URI mismatch: {}",
        schema
    );
    assert_eq!(v["version"].as_str().unwrap(), "2.1.0");
}

#[test]
fn test_sarif_driver_metadata() {
    let tmp = TempDir::new().unwrap();
    save_sarif_report("p", tmp.path(), &[], None).unwrap();
    let v = read_sarif(&tmp, "p");
    let driver = &v["runs"][0]["tool"]["driver"];
    assert_eq!(driver["name"].as_str().unwrap(), "sc2sbom");
    assert!(!driver["version"].as_str().unwrap().is_empty());
}

#[test]
fn test_sarif_rules_deduplication() {
    let tmp = TempDir::new().unwrap();
    let mut findings: Vec<SastFinding> = (0..50)
        .map(|i| make_finding("c", &format!("f{}.c", i), 1, 120))
        .collect();
    findings.push(make_finding("c", "other.c", 2, 78));
    save_sarif_report("p", tmp.path(), &findings, None).unwrap();
    let v = read_sarif(&tmp, "p");
    let rules = v["runs"][0]["tool"]["driver"]["rules"].as_array().unwrap();
    assert_eq!(rules.len(), 2, "expected exactly 2 deduplicated rules");
    let ids: Vec<&str> = rules.iter().map(|r| r["id"].as_str().unwrap()).collect();
    assert!(ids.contains(&"CWE-78"));
    assert!(ids.contains(&"CWE-120"));
}

#[test]
fn test_sarif_rule_fields() {
    let tmp = TempDir::new().unwrap();
    let findings = vec![make_finding("c", "f.c", 1, 120)];
    save_sarif_report("p", tmp.path(), &findings, None).unwrap();
    let v = read_sarif(&tmp, "p");
    let rule = &v["runs"][0]["tool"]["driver"]["rules"][0];
    assert_eq!(rule["id"].as_str().unwrap(), "CWE-120");
    assert!(!rule["name"].as_str().unwrap().is_empty());
    assert_eq!(
        rule["helpUri"].as_str().unwrap(),
        "https://cwe.mitre.org/data/definitions/120.html"
    );
}

#[test]
fn test_sarif_results_no_dedup() {
    let tmp = TempDir::new().unwrap();
    let findings = vec![
        make_finding("a", "src/a.c", 10, 120),
        make_finding("a", "src/a.c", 20, 120),
        make_finding("b", "src/b.c", 30, 78),
    ];
    save_sarif_report("p", tmp.path(), &findings, None).unwrap();
    let v = read_sarif(&tmp, "p");
    let results = v["runs"][0]["results"].as_array().unwrap();
    assert_eq!(results.len(), 3, "expected one result per finding");

    let r0 = &results[0];
    assert_eq!(r0["ruleId"].as_str().unwrap(), "CWE-120");
    assert!(!r0["message"]["text"].as_str().unwrap().is_empty());
    let loc = &r0["locations"][0]["physicalLocation"];
    assert_eq!(loc["artifactLocation"]["uri"].as_str().unwrap(), "src/a.c");
    assert_eq!(loc["region"]["startLine"].as_u64().unwrap(), 10);
}

#[test]
fn test_sarif_empty_findings() {
    let tmp = TempDir::new().unwrap();
    save_sarif_report("empty", tmp.path(), &[], None).unwrap();
    let v = read_sarif(&tmp, "empty");
    assert!(v["runs"][0]["results"].as_array().unwrap().is_empty());
    assert!(v["runs"][0]["tool"]["driver"]["rules"]
        .as_array()
        .unwrap()
        .is_empty());
}
```
