```rust
#[test]
fn test_sarif_custom_path_override() {
    let tmp = TempDir::new().unwrap();
    let custom = tmp.path().join("custom_name.sarif");
    save_sarif_report(
        "ignored_project",
        tmp.path(),
        &[],
        Some(custom.to_str().unwrap()),
    )
    .unwrap();
    assert!(custom.exists(), "custom override path should be written");
    let default = tmp.path().join("ignored_project_static_analysis.sarif");
    assert!(
        !default.exists(),
        "default path must NOT be written when override is set"
    );
}

#[test]
fn test_sarif_custom_path_creates_parent_dir() {
    let tmp = TempDir::new().unwrap();
    let nested = tmp.path().join("subdir").join("nested").join("out.sarif");
    save_sarif_report("p", tmp.path(), &[], Some(nested.to_str().unwrap())).unwrap();
    assert!(nested.exists(), "nested parent dirs should be created");
}

#[test]
fn test_sarif_default_path_when_none() {
    let tmp = TempDir::new().unwrap();
    save_sarif_report("regress", tmp.path(), &[], None).unwrap();
    let path = tmp.path().join("regress_static_analysis.sarif");
    assert!(
        path.exists(),
        "default path must still work when sarif_path is None"
    );
}
```
