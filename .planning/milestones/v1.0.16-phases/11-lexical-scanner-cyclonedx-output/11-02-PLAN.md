    //! Phase 11 (v1.0.16): Pure-Rust lexical CWE scanner.
    //!
    //! Walks component-mapped C/C++ source directories (`ScanContext.component_dirs`)
    //! and reports dangerous-function call sites by file + line. Detection is a
    //! line-level token matcher with a static rule table; no AST, no dataflow.
    //!
    //! Requirements: SCAN-01 (file extensions), SCAN-02 (13 CWEs), SCAN-03
    //! (CWE-134 format-arg heuristic), SCAN-04 (file+line per finding),
    //! SCAN-05 (component-mapped scope).
    //!
    //! All items in this module are gated behind `feature = "internal"`.

    #![cfg(feature = "internal")]

    use crate::util::warn_on_walkdir_err;
    use std::collections::HashMap;
    use std::io::BufRead;
    use std::path::{Path, PathBuf};
    use walkdir::WalkDir;

    /// Single dangerous-function call site detected by the lexical scanner.
    #[derive(Debug, Clone)]
    pub struct SastFinding {
        pub cwe_id: u32,
        pub component_name: String,
        pub component_ecosystem: String,
        /// File path as a string, relative or absolute as discovered by WalkDir.
        pub file_path: String,
        /// 1-indexed line number of the call site.
        pub line: u32,
    }

    /// One CWE rule in the static detection table.
    struct CweRule {
        cwe_id: u32,
        functions: &'static [&'static str],
        /// CWE-134 only: when true, the rule fires only if the format arg
        /// (first arg after the opening paren) is NOT a string literal.
        requires_format_heuristic: bool,
    }

    /// All 13 CWEs from SEED-001 (SCAN-02).
    /// Function lists per Pattern 1 in 11-RESEARCH.md.
    static CWE_RULES: &[CweRule] = &[
        CweRule { cwe_id: 120, functions: &["gets", "strcpy", "strcat", "sprintf", "vsprintf"], requires_format_heuristic: false },
        CweRule { cwe_id: 78,  functions: &["system", "popen", "execl", "execlp", "execle", "execv", "execvp", "execvpe"], requires_format_heuristic: false },
        CweRule { cwe_id: 242, functions: &["gets", "mktemp"], requires_format_heuristic: false },
        CweRule { cwe_id: 327, functions: &["MD5", "MD5_Init", "SHA1", "SHA1_Init", "DES_ecb_encrypt", "EVP_md5", "EVP_sha1"], requires_format_heuristic: false },
        CweRule { cwe_id: 377, functions: &["tmpnam", "tempnam", "mktemp"], requires_format_heuristic: false },
        CweRule { cwe_id: 190, functions: &["malloc", "calloc", "realloc"], requires_format_heuristic: false },
        CweRule { cwe_id: 134, functions: &["printf", "fprintf", "sprintf", "snprintf", "vprintf", "vfprintf", "vsprintf", "vsnprintf", "syslog"], requires_format_heuristic: true },
        CweRule { cwe_id: 22,  functions: &["realpath", "getcwd", "chdir", "open", "fopen"], requires_format_heuristic: false },
        CweRule { cwe_id: 807, functions: &["getenv", "getlogin", "cuserid"], requires_format_heuristic: false },
        CweRule { cwe_id: 362, functions: &["access", "stat", "lstat"], requires_format_heuristic: false },
        CweRule { cwe_id: 367, functions: &["access", "stat"], requires_format_heuristic: false },
        CweRule { cwe_id: 20,  functions: &["atoi", "atol", "atof", "atoll", "strtol", "strtoul"], requires_format_heuristic: false },
        CweRule { cwe_id: 126, functions: &["strlen", "wcslen"], requires_format_heuristic: false },
        CweRule { cwe_id: 676, functions: &["gets", "scanf", "fscanf", "sscanf"], requires_format_heuristic: false },
    ];

    /// Returns true if `after_func` (the slice immediately AFTER the function name)
    /// represents a call whose first argument is a string literal.
    /// Specifically: trim leading `(` and whitespace, then check if the next char is `"`.
    fn format_arg_is_literal(after_func: &str) -> bool {
        let trimmed = after_func.trim_start_matches(|c: char| c == '(' || c.is_whitespace());
        trimmed.starts_with('"')
    }

    /// Find a paren-bound, word-boundary occurrence of `func` in `line`.
    /// Returns the byte offset of the start of the match, or None if not found.
    /// Word boundary: char before match must NOT be ASCII alnum or `_`.
    /// Paren bound: the next non-whitespace char after the function name must be `(`.
    fn find_function_call(line: &str, func: &str) -> Option<usize> {
        let bytes = line.as_bytes();
        let func_bytes = func.as_bytes();
        let mut search_from = 0usize;
        while let Some(rel) = line[search_from..].find(func) {
            let pos = search_from + rel;
            // Check left boundary
            let left_ok = if pos == 0 {
                true
            } else {
                let prev = bytes[pos - 1];
                !(prev.is_ascii_alphanumeric() || prev == b'_')
            };
            if left_ok {
                let after_idx = pos + func_bytes.len();
                if after_idx <= bytes.len() {
                    let after = &line[after_idx..];
                    let trimmed = after.trim_start();
                    if trimmed.starts_with('(') {
                        return Some(pos);
                    }
                }
            }
            search_from = pos + 1;
            if search_from >= line.len() { break; }
        }
        None
    }

    /// Scan one file. Returns empty vec on open or read errors (graceful per project pattern).
    fn scan_file(path: &Path, component_name: &str, component_ecosystem: &str) -> Vec<SastFinding> {
        let file = match std::fs::File::open(path) {
            Ok(f) => f,
            Err(_) => return Vec::new(),
        };
        let reader = std::io::BufReader::new(file);
        let mut findings = Vec::new();

        for (line_idx, line_result) in reader.lines().enumerate() {
            let line = match line_result { Ok(l) => l, Err(_) => continue };
            let line_num = (line_idx + 1) as u32;

            for rule in CWE_RULES {
                for &func in rule.functions {
                    if let Some(pos) = find_function_call(&line, func) {
                        if rule.requires_format_heuristic {
                            let after = &line[pos + func.len()..];
                            if format_arg_is_literal(after) {
                                continue; // safe — string literal format arg
                            }
                        }
                        findings.push(SastFinding {
                            cwe_id: rule.cwe_id,
                            component_name: component_name.to_string(),
                            component_ecosystem: component_ecosystem.to_string(),
                            file_path: path.to_string_lossy().into_owned(),
                            line: line_num,
                        });
                    }
                }
            }
        }
        findings
    }

    fn is_c_cpp_source(path: &Path) -> bool {
        match path.extension().and_then(|e| e.to_str()) {
            Some("c") | Some("h") | Some("cpp") | Some("hpp") | Some("cc") => true,
            _ => false,
        }
    }

    /// Phase 11 entry point. Walks every component-mapped directory and scans
    /// .c/.h/.cpp/.hpp/.cc files for dangerous calls.
    pub fn run_lexical_scanner(
        component_dirs: &HashMap<(String, String), PathBuf>,
    ) -> Vec<SastFinding> {
        let mut all_findings = Vec::new();
        for ((name, ecosystem), dir) in component_dirs.iter() {
            if !dir.exists() {
                continue;
            }
            for entry in WalkDir::new(dir)
                .follow_links(true)
                .into_iter()
                .filter_map(warn_on_walkdir_err)
            {
                if !entry.file_type().is_file() { continue; }
                let p = entry.path();
                if !is_c_cpp_source(p) { continue; }
                all_findings.extend(scan_file(p, name, ecosystem));
            }
        }
        all_findings
    }

    #[cfg(test)]
    mod tests {
        use super::*;

        #[test]
        fn test_format_arg_is_literal_true() {
            assert!(format_arg_is_literal("(\"hello\")"));
            assert!(format_arg_is_literal("(  \"hello\" )"));
        }

        #[test]
        fn test_format_arg_is_literal_false() {
            assert!(!format_arg_is_literal("(user_input)"));
            assert!(!format_arg_is_literal("(buf, \"%s\", x)"));
        }

        #[test]
        fn test_find_function_call_paren_required() {
            assert_eq!(find_function_call("strcpy(d, s);", "strcpy"), Some(0));
            assert_eq!(find_function_call("/* strcpy returns length */", "strcpy"), None);
        }

        #[test]
        fn test_find_function_call_word_boundary() {
            assert_eq!(find_function_call("mystrcpy(x)", "strcpy"), None);
            assert_eq!(find_function_call("abc strcpy(x)", "strcpy"), Some(4));
        }

        #[test]
        fn test_rule_table_has_thirteen_cwes() {
            // SEED-001 (SCAN-02): 13 distinct CWE IDs.
            // The planned table has 14 entries because CWE-242 and CWE-377 each list `mktemp`
            // and other CWEs share function names; SCAN-02 counts DISTINCT CWE IDs, not entries.
            let mut ids: Vec<u32> = CWE_RULES.iter().map(|r| r.cwe_id).collect();
            ids.sort();
            ids.dedup();
            assert_eq!(ids.len(), 13, "SCAN-02 requires 13 distinct CWE IDs in CWE_RULES");
        }
    }

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Filesystem -> scanner | C/C++ source files (potentially adversarial) are read line-by-line. Non-UTF8 bytes and oversize lines must not panic the process. |
| Untrusted source content -> scanner internals | A malicious source file could attempt resource exhaustion via crafted long lines or pathological identifier sequences. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-11-02-01 | DoS | scan_file reading a multi-GB minified .c file | mitigate | `BufRead::lines()` is streaming — memory is bounded by max line length, not file size. Existing project pattern. |
| T-11-02-02 | Tampering | Non-UTF8 bytes in source file cause panic | mitigate | Per-line `Result` is matched with `Err(_) => continue`; no `unwrap` on UTF-8 conversion. `to_string_lossy` used for path. |
| T-11-02-03 | Information Disclosure | Scanner walks outside `dir` via symlinks | accept | `WalkDir::follow_links(true)` is consistent with the existing `scan_directory` policy; this is the project's chosen tradeoff. No new exposure vs. baseline scanner. |
| T-11-02-04 | Spoofing | Crafted file paths embedded in findings could affect downstream consumers | mitigate | bom-ref sanitization is applied in Plan 03 (`/` and `.` -> `-`). file_path values are emitted verbatim only inside JSON properties (not URLs) and are escaped by serde_json. |
| T-11-02-05 | DoS | Pathological line with millions of `strcpy` substrings | accept | Inner `find` loop is O(n*m) per line; worst-case still bounded by line length. No further mitigation justified at this tier (pattern-based scanner). |
</threat_model>
