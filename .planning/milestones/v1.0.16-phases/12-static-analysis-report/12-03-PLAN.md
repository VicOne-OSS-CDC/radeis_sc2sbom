Step C — Insert the SAST section emit. After the closing `}` of the `if check_vulnerabilities && total_vulns > 0 { ... }` block and BEFORE `if summary_only { ... }`, insert this exact block:

```rust
#[cfg(feature = "internal")]
{
    writeln!(output, "## Static Analysis Findings\n")?;
    writeln!(output, "| Component | CWE | Name | Count |")?;
    writeln!(output, "|-----------|-----|------|-------|")?;

    if sast_findings.is_empty() {
        writeln!(output, "| — | — | No static analysis findings detected. | — |")?;
    } else {
        use std::collections::BTreeMap;
        let mut counts: BTreeMap<(String, u32), usize> = BTreeMap::new();
        for f in sast_findings {
            *counts
                .entry((f.component_name.clone(), f.cwe_id))
                .or_insert(0) += 1;
        }
        for ((component, cwe_id), count) in &counts {
            writeln!(
                output,
                "| {} | CWE-{} | {} | {} |",
                component,
                cwe_id,
                cwe_name(*cwe_id),
                count
            )?;
        }
    }
    writeln!(output)?;
}
```
