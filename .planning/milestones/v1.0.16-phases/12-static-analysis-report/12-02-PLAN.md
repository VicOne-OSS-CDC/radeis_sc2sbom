```rust
#[cfg(feature = "internal")]
pub fn save_static_analysis_report(
    project_name: &str,
    out_dir: &Path,
    findings: &[SastFinding],
) -> Result<()> {
    let mut output = String::new();

    // H1 + disclaimer blockquote (D-11)
    writeln!(output, "# Static Analysis Report\n")?;
    writeln!(
        output,
        "> **Note:** Pattern-based — complex data-flow vulnerabilities not covered.\n"
    )?;

    // Summary table (D-04)
    writeln!(output, "## Summary\n")?;
    writeln!(output, "| Component | CWE | Name | Count |")?;
    writeln!(output, "|-----------|-----|------|-------|")?;

    if findings.is_empty() {
        // D-03 zero-findings note row
        writeln!(output, "| — | — | No static analysis findings detected. | — |")?;
    } else {
        // Group by (component, cwe_id) and count. Use BTreeMap for stable order.
        use std::collections::BTreeMap;
        let mut counts: BTreeMap<(String, u32), usize> = BTreeMap::new();
        for f in findings {
            *counts
                .entry((f.component_name.clone(), f.cwe_id))
                .or_insert(0) += 1;
        }
        for ((component, cwe_id), count) in &counts {
            writeln!(
                output,
                "| {} | CWE-{} | {} | {} |",
                component,
                cwe_id,
                cwe_name(*cwe_id),
                count
            )?;
        }
    }
    writeln!(output)?;

    // Findings section (D-05, D-06)
    writeln!(output, "## Findings\n")?;
    if findings.is_empty() {
        writeln!(output, "No static analysis findings detected.\n")?;
    } else {
        // Group: component -> cwe_id -> Vec<&SastFinding>
        use std::collections::BTreeMap;
        let mut grouped: BTreeMap<String, BTreeMap<u32, Vec<&SastFinding>>> = BTreeMap::new();
        for f in findings {
            grouped
                .entry(f.component_name.clone())
                .or_default()
                .entry(f.cwe_id)
                .or_default()
                .push(f);
        }
        for (component, cwes) in &grouped {
            writeln!(output, "## {}\n", component)?;
            for (cwe_id, items) in cwes {
                writeln!(output, "### CWE-{} ({})\n", cwe_id, cwe_name(*cwe_id))?;
                for item in items {
                    writeln!(
                        output,
                        "- {}:{} — {}",
                        item.file_path, item.line, item.function_name
                    )?;
                }
                writeln!(output)?;
            }
        }
    }

    // Write file (D-01, D-02)
    let path = out_dir.join(format!("{}_static_analysis.md", project_name));
    fs::write(&path, output)?;

    // D-10: stderr disclaimer + confirmation
    eprintln!("Pattern-based — complex data-flow vulnerabilities not covered");
    eprintln!("✓ Static analysis report saved to: {}", path.display());

    Ok(())
}
```
